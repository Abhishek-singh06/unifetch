-- Admin check helper (SECURITY DEFINER avoids RLS recursion)
CREATE OR REPLACE FUNCTION public.is_admin()
RETURNS BOOLEAN
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  RETURN EXISTS (
    SELECT 1 FROM public.profiles WHERE id = auth.uid() AND role = 'admin'
  );
END;
$$;

-- Approved student check (Allows 'approved' or 'verified' status, rejects deactivated)
CREATE OR REPLACE FUNCTION public.is_approved_student()
RETURNS BOOLEAN
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  RETURN EXISTS (
    SELECT 1 FROM public.profiles
    WHERE id = auth.uid() 
      AND verification_status IN ('approved', 'verified')
      AND status <> 'deactivated'
  );
END;
$$;

-- Deactivation check trigger function
CREATE OR REPLACE FUNCTION public.check_user_status()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  IF EXISTS (
    SELECT 1 FROM public.profiles
    WHERE id = auth.uid() AND status = 'deactivated'
  ) THEN
    RAISE EXCEPTION 'This account has been deactivated.';
  END IF;
  RETURN NEW;
END;
$$;

-- Protect privileged columns on profiles trigger
DROP TRIGGER IF EXISTS protect_profile_privileged_columns_trg ON public.profiles;

CREATE OR REPLACE FUNCTION public.protect_profile_privileged_columns()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  IF auth.uid() IS NULL THEN
    RETURN NEW;
  END IF;

  IF public.is_admin() THEN
    RETURN NEW;
  END IF;

  IF current_setting('unifetch.allow_credit_payout', true) = 'true' THEN
    RETURN NEW;
  END IF;

  IF NEW.role IS DISTINCT FROM OLD.role
     OR NEW.credits IS DISTINCT FROM OLD.credits THEN
    RAISE EXCEPTION 'Permission denied: role and credits can only be changed by an admin';
  END IF;

  -- Regular users may upload their ID and set status to submitted/id_submitted or reset to pending
  IF NEW.verification_status IS DISTINCT FROM OLD.verification_status
     AND NEW.verification_status NOT IN ('pending', 'submitted', 'id_submitted') THEN
    RAISE EXCEPTION 'Permission denied: only an admin can verify or reject a student ID';
  END IF;

  RETURN NEW;
END;
$$;

CREATE TRIGGER protect_profile_privileged_columns_trg
  BEFORE UPDATE ON public.profiles
  FOR EACH ROW EXECUTE FUNCTION public.protect_profile_privileged_columns();

CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  INSERT INTO public.profiles (
    id, full_name, college, email, verification_status, role, credits, status, created_at, updated_at
  )
  VALUES (
    NEW.id,
    COALESCE(NEW.raw_user_meta_data->>'full_name', ''),
    COALESCE(NEW.raw_user_meta_data->>'college', ''),
    COALESCE(NEW.email, ''),
    'pending',
    'student',
    100,
    'active',
    NOW(),
    NOW()
  )
  ON CONFLICT (id) DO NOTHING;
  RETURN NEW;
END;
$$;

CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();

-- Safely backfill request_otps from legacy pickup_otp if the column exists in package_requests
DO $$
BEGIN
  IF EXISTS (
    SELECT 1 FROM information_schema.columns 
    WHERE table_schema = 'public' 
      AND table_name = 'package_requests' 
      AND column_name = 'pickup_otp'
  ) THEN
    INSERT INTO public.request_otps (request_id, otp)
    SELECT id, pickup_otp FROM public.package_requests
    WHERE pickup_otp IS NOT NULL
    ON CONFLICT (request_id) DO NOTHING;
  END IF;
END;
$$;

CREATE OR REPLACE FUNCTION public.create_package_request(
  p_package_description TEXT,
  p_pickup_location TEXT,
  p_delivery_location TEXT,
  p_pickup_time TIMESTAMPTZ
)
RETURNS UUID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_request_id UUID;
  v_otp TEXT;
  v_requester_credits INTEGER;
BEGIN
  IF auth.uid() IS NULL THEN
    RAISE EXCEPTION 'Not authenticated';
  END IF;

  IF NOT public.is_approved_student() THEN
    RAISE EXCEPTION 'User not approved to create package requests';
  END IF;

  SELECT credits INTO v_requester_credits
  FROM public.profiles
  WHERE id = auth.uid()
  FOR UPDATE;

  IF v_requester_credits < 50 THEN
    RAISE EXCEPTION 'Insufficient credits. You need at least 50 credits to make a request.';
  END IF;

  UPDATE public.profiles
  SET credits = credits - 50, updated_at = NOW()
  WHERE id = auth.uid();

  INSERT INTO public.package_requests (
    requester_id, package_description, pickup_location, delivery_location, pickup_time, status
  )
  VALUES (
    auth.uid(), p_package_description, p_pickup_location, p_delivery_location, p_pickup_time, 'pending'
  )
  RETURNING id INTO v_request_id;

  v_otp := LPAD(FLOOR(RANDOM() * 1000000)::TEXT, 6, '0');

  INSERT INTO public.request_otps (request_id, otp)
  VALUES (v_request_id, v_otp);

  RETURN v_request_id;
END;
$$;

DROP FUNCTION IF EXISTS public.claim_package_request(uuid);

CREATE OR REPLACE FUNCTION public.claim_package_request(
  p_request_id UUID
)
RETURNS BOOLEAN
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_current_status TEXT;
  v_requester_id UUID;
BEGIN
  IF auth.uid() IS NULL THEN
    RAISE EXCEPTION 'Not authenticated';
  END IF;

  IF NOT public.is_approved_student() THEN
    RAISE EXCEPTION 'User not approved to claim deliveries';
  END IF;

  SELECT status, requester_id INTO v_current_status, v_requester_id
  FROM public.package_requests
  WHERE id = p_request_id
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Request not found';
  END IF;

  IF v_current_status <> 'pending' THEN
    RAISE EXCEPTION 'Request is no longer available';
  END IF;

  IF v_requester_id = auth.uid() THEN
    RAISE EXCEPTION 'You cannot claim your own request';
  END IF;

  UPDATE public.package_requests
  SET
    carrier_id = auth.uid(),
    status = 'matched',
    updated_at = NOW()
  WHERE id = p_request_id;

  RETURN TRUE;
END;
$$;

DROP FUNCTION IF EXISTS public.get_my_request_otp(uuid);

CREATE OR REPLACE FUNCTION public.get_my_request_otp(
  p_request_id UUID
)
RETURNS TEXT
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_otp TEXT;
BEGIN
  IF auth.uid() IS NULL THEN
    RAISE EXCEPTION 'Not authenticated';
  END IF;

  SELECT ro.otp INTO v_otp
  FROM public.request_otps ro
  JOIN public.package_requests pr ON pr.id = ro.request_id
  WHERE ro.request_id = p_request_id
    AND pr.requester_id = auth.uid();

  IF NOT FOUND THEN
    RAISE EXCEPTION 'OTP not found or access denied';
  END IF;

  RETURN v_otp;
END;
$$;

DROP FUNCTION IF EXISTS public.verify_package_otp(uuid, text);
DROP FUNCTION IF EXISTS public.verify_package_otp(uuid, varchar);

CREATE OR REPLACE FUNCTION public.verify_package_otp(
  p_request_id UUID,
  p_otp TEXT
)
RETURNS BOOLEAN
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_stored_otp TEXT;
  v_carrier_id UUID;
  v_status TEXT;
  v_requester_id UUID;
BEGIN
  IF auth.uid() IS NULL THEN
    RAISE EXCEPTION 'Not authenticated';
  END IF;

  IF NOT public.is_approved_student() THEN
    RAISE EXCEPTION 'User not approved for package requests';
  END IF;

  SELECT pr.carrier_id, pr.status, pr.requester_id, ro.otp
  INTO v_carrier_id, v_status, v_requester_id, v_stored_otp
  FROM public.package_requests pr
  LEFT JOIN public.request_otps ro ON ro.request_id = pr.id
  WHERE pr.id = p_request_id;

  IF NOT FOUND THEN
    RETURN FALSE;
  END IF;

  IF v_carrier_id IS DISTINCT FROM auth.uid() THEN
    RETURN FALSE;
  END IF;

  IF v_status <> 'matched' THEN
    RETURN FALSE;
  END IF;

  IF v_stored_otp IS NOT NULL AND v_stored_otp = p_otp THEN
    UPDATE public.package_requests
    SET
      status = 'delivered',
      otp_verified = TRUE,
      delivered_at = NOW(),
      updated_at = NOW()
    WHERE id = p_request_id;

    PERFORM set_config('unifetch.allow_credit_payout', 'true', false);
    UPDATE public.profiles
    SET credits = credits + 35, updated_at = NOW()
    WHERE id = v_carrier_id;
    PERFORM set_config('unifetch.allow_credit_payout', 'false', false);

    RETURN TRUE;
  END IF;

  RETURN FALSE;
END;
$$;

CREATE OR REPLACE FUNCTION public.buy_credits(
  p_credits INTEGER,
  p_payment_reference TEXT,
  p_payment_proof_url TEXT
)
RETURNS UUID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_purchase_id UUID;
  v_amount NUMERIC(10,2);
BEGIN
  IF auth.uid() IS NULL THEN
    RAISE EXCEPTION 'Not authenticated';
  END IF;

  IF p_credits = 100 THEN
    v_amount := 99.00;
  ELSIF p_credits = 250 THEN
    v_amount := 229.00;
  ELSIF p_credits = 600 THEN
    v_amount := 499.00;
  ELSIF p_credits = 1500 THEN
    v_amount := 999.00;
  ELSE
    RAISE EXCEPTION 'Invalid credit package: %', p_credits;
  END IF;

  INSERT INTO public.credit_purchases (
    user_id, credits, amount_inr, payment_reference, payment_proof_url, status
  )
  VALUES (
    auth.uid(), p_credits, v_amount, p_payment_reference, p_payment_proof_url, 'pending'
  )
  RETURNING id INTO v_purchase_id;

  RETURN v_purchase_id;
END;
$$;

DROP FUNCTION IF EXISTS public.approve_credit_purchase(uuid);

CREATE OR REPLACE FUNCTION public.approve_credit_purchase(
  p_purchase_id UUID
)
RETURNS BOOLEAN
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_user_id UUID;
  v_credits INTEGER;
  v_status TEXT;
BEGIN
  IF auth.uid() IS NULL OR NOT public.is_admin() THEN
    RAISE EXCEPTION 'Permission denied: only admins may approve credit purchases';
  END IF;

  SELECT user_id, credits, status
  INTO v_user_id, v_credits, v_status
  FROM public.credit_purchases
  WHERE id = p_purchase_id
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Purchase request not found';
  END IF;

  IF v_status <> 'pending' THEN
    RAISE EXCEPTION 'Purchase request is already processed (%)', v_status;
  END IF;

  UPDATE public.credit_purchases
  SET
    status = 'completed',
    reviewed_at = NOW(),
    reviewed_by = auth.uid()
  WHERE id = p_purchase_id;

  UPDATE public.profiles
  SET
    credits = credits + v_credits,
    updated_at = NOW()
  WHERE id = v_user_id;

  RETURN TRUE;
END;
$$;

DROP FUNCTION IF EXISTS public.reject_credit_purchase(uuid, text);

CREATE OR REPLACE FUNCTION public.reject_credit_purchase(
  p_purchase_id UUID,
  p_rejection_reason TEXT
)
RETURNS BOOLEAN
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_status TEXT;
BEGIN
  IF auth.uid() IS NULL OR NOT public.is_admin() THEN
    RAISE EXCEPTION 'Permission denied: only admins may reject credit purchases';
  END IF;

  SELECT status INTO v_status
  FROM public.credit_purchases
  WHERE id = p_purchase_id
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Purchase request not found';
  END IF;

  IF v_status <> 'pending' THEN
    RAISE EXCEPTION 'Purchase request is already processed (%)', v_status;
  END IF;

  UPDATE public.credit_purchases
  SET
    status = 'rejected',
    rejection_reason = p_rejection_reason,
    reviewed_at = NOW(),
    reviewed_by = auth.uid()
  WHERE id = p_purchase_id;

  RETURN TRUE;
END;
$$;

DROP TRIGGER IF EXISTS protect_outside_request_transitions_trg ON public.outside_requests;

CREATE OR REPLACE FUNCTION public.protect_outside_request_transitions()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  IF auth.uid() IS NULL THEN
    RAISE EXCEPTION 'Not authenticated';
  END IF;

  IF NEW.id <> OLD.id OR NEW.requester_id <> OLD.requester_id THEN
    RAISE EXCEPTION 'Invalid operation: id and requester_id cannot be changed';
  END IF;

  IF auth.uid() = OLD.carrier_id AND auth.uid() <> OLD.requester_id THEN
    IF NEW.status = 'ACCEPTED' AND OLD.status <> 'ACCEPTED' THEN
      RAISE EXCEPTION 'Only the requester can accept a reward proposal';
    END IF;

    IF NEW.payment_status = 'Sent' AND OLD.payment_status <> 'Sent' THEN
      RAISE EXCEPTION 'Only the requester can mark payment as Sent';
    END IF;
  END IF;

  IF auth.uid() = OLD.requester_id THEN
    IF NEW.payment_qr_url IS DISTINCT FROM OLD.payment_qr_url THEN
      RAISE EXCEPTION 'Only the carrier can upload payment details';
    END IF;

    IF NEW.payment_status IN ('Confirmed', 'Received') AND OLD.payment_status NOT IN ('Confirmed', 'Received') THEN
      RAISE EXCEPTION 'Only the carrier can confirm payment receipt';
    END IF;

    IF (NEW.status IN ('IN PROGRESS', 'TASK_STARTED', 'COMPLETED')) AND NEW.status <> OLD.status THEN
      RAISE EXCEPTION 'Only the carrier can advance task execution status';
    END IF;
  END IF;

  RETURN NEW;
END;
$$;

CREATE TRIGGER protect_outside_request_transitions_trg
  BEFORE UPDATE ON public.outside_requests
  FOR EACH ROW EXECUTE FUNCTION public.protect_outside_request_transitions();

DROP TRIGGER IF EXISTS trg_outside_status_change ON public.outside_requests;

CREATE OR REPLACE FUNCTION public.handle_outside_request_status_change()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_carrier_name TEXT;
BEGIN
  -- Carrier claims the request
  IF OLD.carrier_id IS NULL AND NEW.carrier_id IS NOT NULL THEN
    SELECT full_name INTO v_carrier_name FROM public.profiles WHERE id = NEW.carrier_id;
    INSERT INTO public.outside_messages (request_id, sender_id, content, is_system)
    VALUES (NEW.id, NULL, COALESCE(v_carrier_name, 'Carrier') || ' joined negotiation.', TRUE);
  END IF;

  -- Status transitions
  IF OLD.status IS DISTINCT FROM NEW.status THEN
    IF NEW.status = 'ACCEPTED' THEN
      INSERT INTO public.outside_messages (request_id, sender_id, content, is_system)
      VALUES (NEW.id, NULL, 'Task reward agreed! Status advanced to ACCEPTED.', TRUE);
    ELSIF NEW.status = 'PAYMENT PENDING' THEN
      INSERT INTO public.outside_messages (request_id, sender_id, content, is_system)
      VALUES (NEW.id, NULL, 'Payment details uploaded. Status is now PAYMENT PENDING.', TRUE);
    ELSIF NEW.status = 'PAID' THEN
      INSERT INTO public.outside_messages (request_id, sender_id, content, is_system)
      VALUES (NEW.id, NULL, 'Payment confirmed! Task status advanced to PAID.', TRUE);
    ELSIF NEW.status IN ('IN PROGRESS', 'TASK_STARTED') THEN
      INSERT INTO public.outside_messages (request_id, sender_id, content, is_system)
      VALUES (NEW.id, NULL, 'Task is now IN PROGRESS.', TRUE);
    ELSIF NEW.status = 'COMPLETED' THEN
      INSERT INTO public.outside_messages (request_id, sender_id, content, is_system)
      VALUES (NEW.id, NULL, 'Task marked as COMPLETED! 🎉', TRUE);
    ELSIF NEW.status = 'CANCELLED' THEN
      INSERT INTO public.outside_messages (request_id, sender_id, content, is_system)
      VALUES (NEW.id, NULL, 'Task was cancelled.', TRUE);
    END IF;
  END IF;

  -- Payment status changes
  IF OLD.payment_status IS DISTINCT FROM NEW.payment_status THEN
    IF NEW.payment_status = 'Sent' THEN
      INSERT INTO public.outside_messages (request_id, sender_id, content, is_system)
      VALUES (NEW.id, NULL, 'Payment marked as Sent. Waiting for Carrier confirmation.', TRUE);
    END IF;
  END IF;

  RETURN NEW;
END;
$$;

CREATE TRIGGER trg_outside_status_change
  AFTER UPDATE OF status, payment_status, carrier_id ON public.outside_requests
  FOR EACH ROW EXECUTE FUNCTION public.handle_outside_request_status_change();

-- Notification trigger for Campus requests
DROP TRIGGER IF EXISTS trg_package_request_notifications ON public.package_requests;

CREATE OR REPLACE FUNCTION public.trg_package_request_notifications()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  IF TG_OP = 'INSERT' THEN
    INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
    VALUES (
      NEW.requester_id, 'REQUEST_CREATED', 'Campus Request Created',
      'Your campus request has been created.', '/requests', 'campus_created', NEW.id
    ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;
  END IF;

  IF TG_OP = 'UPDATE' THEN
    IF OLD.status = 'pending' AND NEW.status = 'matched' AND NEW.carrier_id IS NOT NULL THEN
      INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
      VALUES (
        NEW.requester_id, 'REQUEST_ACCEPTED', 'Campus Request Accepted',
        'Someone accepted your campus delivery request.', '/requests', 'campus_accepted', NEW.id
      ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;
    END IF;

    IF OLD.status <> 'delivered' AND NEW.status = 'delivered' AND NEW.otp_verified = TRUE THEN
      INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
      VALUES (
        NEW.requester_id, 'TASK_COMPLETED', 'Campus Delivery Complete',
        'Your campus delivery is complete.', '/requests', 'campus_completed_requester', NEW.id
      ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;

      IF NEW.carrier_id IS NOT NULL THEN
        INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
        VALUES (
          NEW.carrier_id, 'CARRIER_REWARD', 'Credits Earned!',
          'You earned 35 Credits.', '/carry', 'campus_completed_carrier', NEW.id
        ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;
      END IF;
    END IF;
  END IF;

  RETURN NEW;
END;
$$;

CREATE TRIGGER trg_package_request_notifications
  AFTER INSERT OR UPDATE ON public.package_requests
  FOR EACH ROW EXECUTE FUNCTION public.trg_package_request_notifications();

-- Notification trigger for Outside tasks
DROP TRIGGER IF EXISTS trg_outside_request_notifications ON public.outside_requests;

CREATE OR REPLACE FUNCTION public.trg_outside_request_notifications()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_carrier_name TEXT;
BEGIN
  IF TG_OP = 'UPDATE' THEN
    IF OLD.status = 'OPEN' AND NEW.status = 'NEGOTIATING' AND NEW.carrier_id IS NOT NULL THEN
      SELECT full_name INTO v_carrier_name FROM public.profiles WHERE id = NEW.carrier_id;
      INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
      VALUES (
        NEW.requester_id, 'REQUEST_ACCEPTED', 'Outside Task Claimed',
        COALESCE(v_carrier_name, 'A student') || ' accepted your request.', '/outside/' || NEW.id, 'outside_claimed', NEW.id
      ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;
    END IF;

    IF OLD.status = 'NEGOTIATING' AND NEW.status = 'ACCEPTED' AND NEW.final_reward IS NOT NULL THEN
      INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
      VALUES (
        NEW.carrier_id, 'REWARD_AGREED', 'Reward Agreed',
        'Reward agreed: ₹' || NEW.final_reward || '.', '/outside/' || NEW.id, 'outside_agreed_carrier', NEW.id
      ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;

      INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
      VALUES (
        NEW.requester_id, 'REWARD_AGREED', 'Reward Agreed',
        'Reward agreed: ₹' || NEW.final_reward || '.', '/outside/' || NEW.id, 'outside_agreed_requester', NEW.id
      ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;
    END IF;

    IF (OLD.payment_qr_url IS NULL OR OLD.payment_qr_url = '') AND NEW.payment_qr_url IS NOT NULL AND NEW.payment_qr_url <> '' THEN
      SELECT full_name INTO v_carrier_name FROM public.profiles WHERE id = NEW.carrier_id;
      INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
      VALUES (
        NEW.requester_id, 'PAYMENT_DETAILS_SHARED', 'Payment Details Shared',
        'Payment details shared by ' || COALESCE(v_carrier_name, 'carrier') || '.', '/outside/' || NEW.id, 'outside_details_shared', NEW.id
      ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;
    END IF;

    IF OLD.payment_status = 'Pending' AND NEW.payment_status = 'Sent' THEN
      INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
      VALUES (
        NEW.carrier_id, 'PAYMENT_SENT', 'Payment Marked Sent',
        'Payment marked as sent. Please verify receipt.', '/outside/' || NEW.id, 'outside_payment_sent', NEW.id
      ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;
    END IF;

    IF (OLD.payment_status NOT IN ('Confirmed', 'Received')) AND (NEW.payment_status IN ('Confirmed', 'Received')) THEN
      INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
      VALUES (
        NEW.requester_id, 'PAYMENT_CONFIRMED', 'Payment Confirmed',
        'Carrier confirmed payment receipt.', '/outside/' || NEW.id, 'outside_payment_received', NEW.id
      ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;
    END IF;

    IF (OLD.status NOT IN ('IN PROGRESS', 'TASK_STARTED')) AND (NEW.status IN ('IN PROGRESS', 'TASK_STARTED')) THEN
      INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
      VALUES (
        NEW.requester_id, 'TASK_STARTED', 'Task In Progress',
        'Carrier is currently fulfilling your request.', '/outside/' || NEW.id, 'outside_task_started', NEW.id
      ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;
    END IF;

    IF OLD.status <> 'COMPLETED' AND NEW.status = 'COMPLETED' THEN
      INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
      VALUES (
        NEW.requester_id, 'TASK_COMPLETED', 'Outside Delivery Complete',
        'Your outside delivery has been completed.', '/outside/' || NEW.id, 'outside_completed_requester', NEW.id
      ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;

      INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
      VALUES (
        NEW.carrier_id, 'TASK_COMPLETED', 'Task Complete',
        'Outside task completed.', '/outside/' || NEW.id, 'outside_completed_carrier', NEW.id
      ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;
    END IF;

    IF OLD.status <> 'CANCELLED' AND NEW.status = 'CANCELLED' THEN
      IF auth.uid() = NEW.requester_id AND NEW.carrier_id IS NOT NULL THEN
        INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
        VALUES (
          NEW.carrier_id, 'REQUEST_CANCELLED', 'Request Cancelled',
          'The requester cancelled this request.', '/outside/' || NEW.id, 'outside_cancelled_carrier', NEW.id
        ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;
      ELSIF auth.uid() = NEW.carrier_id THEN
        INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
        VALUES (
          NEW.requester_id, 'REQUEST_CANCELLED', 'Request Cancelled',
          'The carrier cancelled this request.', '/outside/' || NEW.id, 'outside_cancelled_requester', NEW.id
        ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;
      END IF;
    END IF;
  END IF;

  RETURN NEW;
END;
$$;

CREATE TRIGGER trg_outside_request_notifications
  AFTER UPDATE ON public.outside_requests
  FOR EACH ROW EXECUTE FUNCTION public.trg_outside_request_notifications();

-- Notification trigger for Outside Messages
DROP TRIGGER IF EXISTS trg_outside_message_notifications ON public.outside_messages;

CREATE OR REPLACE FUNCTION public.trg_outside_message_notifications()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_recipient_id UUID;
  v_sender_name TEXT;
  v_req RECORD;
BEGIN
  IF NEW.is_system = FALSE THEN
    SELECT requester_id, carrier_id INTO v_req
    FROM public.outside_requests WHERE id = NEW.request_id;

    IF FOUND THEN
      IF NEW.sender_id = v_req.requester_id THEN
        v_recipient_id := v_req.carrier_id;
      ELSE
        v_recipient_id := v_req.requester_id;
      END IF;

      IF v_recipient_id IS NOT NULL THEN
        SELECT full_name INTO v_sender_name FROM public.profiles WHERE id = NEW.sender_id;

        IF NEW.proposal_amount IS NOT NULL THEN
          INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
          VALUES (
            v_recipient_id, 'REWARD_PROPOSAL', 'New Reward Proposal',
            COALESCE(v_sender_name, 'Student') || ' proposed a cash reward of ₹' || NEW.proposal_amount || '.',
            '/outside/' || NEW.request_id, 'outside_proposal_' || NEW.id, NEW.id
          ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;
        ELSE
          INSERT INTO public.notifications (user_id, type, title, message, link, source_type, source_id)
          VALUES (
            v_recipient_id, 'NEW_MESSAGE', 'New Message',
            COALESCE(v_sender_name, 'Student') || ': ' || substring(NEW.content from 1 for 60),
            '/outside/' || NEW.request_id, 'outside_msg_' || NEW.id, NEW.id
          ) ON CONFLICT (user_id, source_type, source_id) DO NOTHING;
        END IF;
      END IF;
    END IF;
  END IF;

  RETURN NEW;
END;
$$;

CREATE TRIGGER trg_outside_message_notifications

  ON storage.objects FOR SELECT
  USING (bucket_id = 'payment-settings');

DROP POLICY IF EXISTS "Admin Manage Payment Settings Storage" ON storage.objects;
CREATE POLICY "Admin Manage Payment Settings Storage"
  ON storage.objects FOR ALL
  USING (bucket_id = 'payment-settings' AND (SELECT public.is_admin()));

-- 4. Outside Delivery Storage (QR / proof uploads)
INSERT INTO storage.buckets (id, name, public)
VALUES ('outside-delivery', 'outside-delivery', true)
ON CONFLICT (id) DO UPDATE SET public = true;

DROP POLICY IF EXISTS "Authenticated users can upload outside delivery media" ON storage.objects;
CREATE POLICY "Authenticated users can upload outside delivery media"
  ON storage.objects FOR INSERT
  WITH CHECK (
    bucket_id = 'outside-delivery' AND
    auth.role() = 'authenticated'
  );

DROP POLICY IF EXISTS "Anyone can view outside delivery media" ON storage.objects;
CREATE POLICY "Anyone can view outside delivery media"
  ON storage.objects FOR SELECT
  USING (bucket_id = 'outside-delivery');
