-- 1. Verify the check constraint on profiles
SELECT 
  conname AS constraint_name,
  pg_get_constraintdef(c.oid) AS constraint_definition
FROM pg_constraint c
JOIN pg_class t ON c.conrelid = t.oid
JOIN pg_namespace n ON t.relnamespace = n.oid
WHERE n.nspname = 'public' 
  AND t.relname = 'profiles'
  AND c.conname = 'profiles_verification_status_check';

-- Expected Output:
-- CHECK (verification_status = ANY (ARRAY['pending'::text, 'id_submitted'::text, 'approved'::text, 'rejected'::text, 'verified'::text, 'submitted'::text]))


-- 2. Verify is_approved_student() definition and security settings
SELECT 
  p.proname AS function_name,
  pg_get_functiondef(p.oid) AS definition,
  p.prosecdef AS is_security_definer,
  l.lanname AS language
FROM pg_proc p
JOIN pg_namespace n ON p.pronamespace = n.oid
JOIN pg_language l ON p.prolang = l.oid
WHERE n.nspname = 'public' 
  AND p.proname = 'is_approved_student';

-- Expected Output:
-- is_approved_student includes 'approved', 'verified', 'id_submitted', role = 'admin', and status <> 'deactivated'


-- 3. Verify protect_profile_privileged_columns() trigger and function
SELECT 
  tgname AS trigger_name,
  tgenabled AS is_enabled,
  pg_get_triggerdef(t.oid) AS trigger_definition
FROM pg_trigger t
JOIN pg_class c ON t.tgrelid = c.oid
JOIN pg_namespace n ON c.relnamespace = n.oid
WHERE n.nspname = 'public' 
  AND c.relname = 'profiles'
  AND t.tgname = 'protect_profile_privileged_columns_trg';

-- Expected Output:
-- Trigger active BEFORE UPDATE on public.profiles


-- 4. Verify existing profile data count and integrity (no profiles deleted)
SELECT 
  verification_status,
  status,
  role,
  COUNT(*) AS profile_count
FROM public.profiles
GROUP BY verification_status, status, role
ORDER BY profile_count DESC;

-- Expected Output:
-- All existing profiles remain intact (total count 52+).
