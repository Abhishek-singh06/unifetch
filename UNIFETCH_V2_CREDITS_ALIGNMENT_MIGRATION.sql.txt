-- ==============================================================================
-- UNIFETCH V2 - CREDITS ALIGNMENT MIGRATION
-- Fixes relation column mismatch (amount vs amount_inr) in credit_purchases
-- and updates buy_credits() function to populate both columns safely.
-- Idempotent, Non-destructive, Zero data loss.
-- ==============================================================================

-- 1. Ensure amount_inr exists in credit_purchases
ALTER TABLE public.credit_purchases ADD COLUMN IF NOT EXISTS amount_inr NUMERIC(10,2) DEFAULT NULL;

-- 2. Backfill existing rows so both columns are synchronized
UPDATE public.credit_purchases SET amount_inr = amount WHERE amount_inr IS NULL AND amount IS NOT NULL;
UPDATE public.credit_purchases SET amount = amount_inr WHERE amount IS NULL AND amount_inr IS NOT NULL;

-- 3. Replace buy_credits() RPC to insert into both amount and amount_inr
DROP FUNCTION IF EXISTS public.buy_credits(integer, text, text);
DROP FUNCTION IF EXISTS public.buy_credits(integer, varchar, varchar);

CREATE OR REPLACE FUNCTION public.buy_credits(
  p_credits INTEGER,
  p_payment_reference TEXT,
  p_payment_proof_url TEXT DEFAULT NULL
)
RETURNS UUID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_user_id UUID;
  v_purchase_id UUID;
  v_amount_inr NUMERIC(10,2);
BEGIN
  IF auth.uid() IS NULL THEN
    RAISE EXCEPTION 'Not authenticated';
  END IF;

  v_user_id := auth.uid();
  v_amount_inr := p_credits::NUMERIC(10,2);

  INSERT INTO public.credit_purchases (
    user_id, credits, amount, amount_inr, payment_reference, payment_proof_url, status
  ) VALUES (
    v_user_id, p_credits, v_amount_inr, v_amount_inr, p_payment_reference, p_payment_proof_url, 'pending'
  )
  RETURNING id INTO v_purchase_id;

  RETURN v_purchase_id;
END;
$$;

GRANT EXECUTE ON FUNCTION public.buy_credits(integer, text, text) TO authenticated;
